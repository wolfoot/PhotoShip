cmake_minimum_required(VERSION 3.21)
project(PixelStudio VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
add_library(editor_core src/document.cpp src/project.cpp third_party/compositor/WandPixels.c)
target_include_directories(editor_core PUBLIC src PRIVATE third_party/compositor)
target_link_libraries(editor_core PUBLIC Qt6::Widgets)
if(UNIX)
  target_link_libraries(editor_core PRIVATE m)
endif()
add_executable(pixelstudio WIN32 src/main.cpp src/canvas.cpp src/window.cpp)
target_link_libraries(pixelstudio PRIVATE editor_core)
if(MSVC)
  target_compile_options(editor_core PRIVATE /W4 /utf-8)
  target_compile_options(pixelstudio PRIVATE /W4 /utf-8)
else()
  target_compile_options(editor_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(pixelstudio PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(editor_tests tests/editor_tests.cpp src/canvas.cpp src/window.cpp)
  target_link_libraries(editor_tests PRIVATE editor_core)
  add_test(NAME editor_tests COMMAND editor_tests)
  set_tests_properties(editor_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
include(GNUInstallDirs)
install(TARGETS pixelstudio RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES third_party/compositor/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/pixelstudio RENAME Compositor-LICENSE)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/pixelstudio)
install(DIRECTORY packaging/qt-licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/pixelstudio/licenses)
if(UNIX AND NOT APPLE)
  install(FILES packaging/pixelstudio.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES packaging/pixelstudio.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
  set(CPACK_GENERATOR DEB)
  set(CPACK_DEBIAN_PACKAGE_SECTION graphics)
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6widgets6 (>= 6.2), libqt6gui6 (>= 6.2), libqt6core6 (>= 6.2), qt6-qpa-plugins")
  set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
endif()
set(CPACK_PACKAGE_CONTACT "Pixel Studio maintainers")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Layer-based desktop image editor")
set(CPACK_PACKAGE_NAME pixelstudio)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
include(CPack)
