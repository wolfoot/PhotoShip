cmake_minimum_required(VERSION 3.21)
project(PhotoShip VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(ZLIB REQUIRED)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Concurrent)
add_library(editor_core src/i18n.cpp src/resources.qrc src/document.cpp src/advanced.cpp src/project.cpp src/psd.cpp src/persistence.cpp third_party/compositor/WandPixels.c)
target_include_directories(editor_core PUBLIC src PRIVATE third_party/compositor)
target_link_libraries(editor_core PUBLIC Qt6::Widgets Qt6::Concurrent ZLIB::ZLIB)
if(UNIX)
  target_link_libraries(editor_core PRIVATE m)
endif()
add_executable(photoship WIN32 src/main.cpp src/canvas.cpp src/window.cpp)
target_link_libraries(photoship PRIVATE editor_core)
if(WIN32)
  enable_language(RC)
  target_sources(photoship PRIVATE packaging/windows.rc)
  target_include_directories(photoship PRIVATE packaging)
endif()
if(MSVC)
  target_compile_options(editor_core PRIVATE /W4 /utf-8)
  target_compile_options(photoship PRIVATE /W4 /utf-8)
else()
  target_compile_options(editor_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(photoship PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(i18n_tests tests/i18n_tests.cpp src/canvas.cpp src/window.cpp)
  target_link_libraries(i18n_tests PRIVATE editor_core)
  add_test(NAME i18n_tests COMMAND i18n_tests)
  set_tests_properties(i18n_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(editor_tests tests/editor_tests.cpp src/canvas.cpp src/window.cpp)
  target_link_libraries(editor_tests PRIVATE editor_core)
  add_executable(v2_tests tests/v2_tests.cpp src/canvas.cpp src/window.cpp)
  target_link_libraries(v2_tests PRIVATE editor_core)
  add_test(NAME v2_tests COMMAND v2_tests)
  set_tests_properties(v2_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME editor_tests COMMAND editor_tests)
  set_tests_properties(editor_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(MSVC)
    foreach(target editor_tests v2_tests i18n_tests)
      target_compile_options(${target} PRIVATE /utf-8)
    endforeach()
  endif()
endif()
include(GNUInstallDirs)
install(TARGETS photoship RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES third_party/compositor/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/photoship RENAME Compositor-LICENSE)
install(FILES packaging/zlib-LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/photoship/licenses)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/photoship)
install(DIRECTORY packaging/qt-licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/photoship/licenses)
if(UNIX AND NOT APPLE)
  install(FILES packaging/photoship.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES packaging/photoship.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
  foreach(size 16 32 48 64 128 256 512)
    install(FILES packaging/icons/photoship-${size}.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps RENAME photoship.png)
  endforeach()
  set(CPACK_GENERATOR DEB)
  set(CPACK_DEBIAN_PACKAGE_SECTION graphics)
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6widgets6 (>= 6.2), libqt6gui6 (>= 6.2), libqt6core6 (>= 6.2), libqt6concurrent6 (>= 6.2), zlib1g, qt6-qpa-plugins")
  set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "fonts-noto-cjk")
  set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
endif()
set(CPACK_PACKAGE_CONTACT "PhotoShip maintainers")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Layer-based desktop image editor")
set(CPACK_PACKAGE_NAME photoship)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
include(CPack)
